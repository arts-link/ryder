{{- /* Schema.org JSON-LD. Every block on this page is built as a Hugo `dict` and rendered with `jsonify`. That is deliberate and load-bearing: the previous version of this file hand-wrote JSON as template text, which meant a stray comment, a trailing comma, or an unquoted value produced a block that no consumer could parse — and nothing in the build ever said so. Structured data failing silently is the single worst failure mode in this file, so the format is now produced by a serializer rather than by careful typing. Rule for anyone editing this file: never write a `{`, a `"` or a `,` that is meant to reach the output. Build a dict, `merge` optional keys in with `with`, and hand it to `jsonify`. Emitted per page: • home — WebPage, plus the site-wide entity (see params.schema.type) • non-home — BlogPosting, plus a single BreadcrumbList for the section path, on pages whose trail has more than just Home. To ADD types (MusicGroup, Event, Product, …) do NOT override this file — overriding it silently drops everything above. Use the no-op `head/schema-extra.html` hook instead, which is called right after this partial and shadows cleanly from a site's own layouts/. Config: [params.schema] type = "Organization" # or MusicGroup, Person, LocalBusiness, … */ -}} {{- $featured := partial "common-partials/opengraph/get-featured-image.html" . -}} {{- /* ── shared entity fragments ──────────────────────────────────────── */ -}} {{- $author := site.Params.author | default dict -}} {{- $authorName := $author.name | default site.Title -}} {{- $authorEmail := $author.email | default "" -}} {{- $logo := "" -}} {{- with site.Params.logo_png -}}{{- $logo = . | absURL -}}{{- end -}} {{- $person := dict "@type" "Person" "name" $authorName "url" site.BaseURL -}} {{- with $authorEmail -}}{{- $person = merge $person (dict "email" .) -}}{{- end -}} {{- $publisher := dict "@type" "Organization" "name" $authorName "url" site.BaseURL -}} {{- with $authorEmail -}}{{- $publisher = merge $publisher (dict "email" .) -}}{{- end -}} {{- with $logo -}}{{- $publisher = merge $publisher (dict "logo" .) -}}{{- end -}} {{- /* Keys every page-level block shares. */ -}} {{- $common := dict "isFamilyFriendly" true "copyrightHolder" $authorName "accountablePerson" $person "author" $person "creator" $person "publisher" $publisher -}} {{- with site.LanguageCode -}}{{- $common = merge $common (dict "inLanguage" .) -}}{{- end -}} {{- if not .PublishDate.IsZero -}} {{- $common = merge $common (dict "copyrightYear" (.PublishDate.Format "2006")) -}} {{- end -}} {{- with $featured -}}{{- $common = merge $common (dict "image" .Permalink) -}}{{- end -}} {{- /* ── page-level block ─────────────────────────────────────────────── */ -}} {{- /* .IsHome is the only homepage test used in this file. The previous version mixed `eq .Title .Site.Title` and `ne .Title .Site.Title` in with it, which handed homepage schema to any page that happened to share the site's title. */ -}} {{- if .IsHome -}} {{- $page := merge $common (dict "@context" "https://schema.org" "@type" "WebPage" "name" .Title "url" site.BaseURL ) -}} {{- else -}} {{- /* description: front matter, then summary, then the site description. */ -}} {{- $description := "" -}} {{- with .Description -}} {{- $description = . | plainify | htmlUnescape | chomp -}} {{- else -}} {{- with .Summary -}} {{- $description = . | plainify | htmlUnescape | chomp -}} {{- else -}} {{- with site.Params.description -}}{{- $description = . | plainify | chomp -}}{{- end -}} {{- end -}} {{- end -}} {{- $page := merge $common (dict "@context" "https://schema.org" "@type" "BlogPosting" "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "headline" (.Title | truncate 110) "url" .Permalink ) -}} {{- /* Dates as RFC 3339. Go's default time rendering ("2006-01-02 15:04:05 -0700 MST") is not a valid schema.org Date. */ -}} {{- if not .PublishDate.IsZero -}} {{- $page = merge $page (dict "datePublished" (.PublishDate.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- if not .Lastmod.IsZero -}} {{- $page = merge $page (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- with $description -}}{{- $page = merge $page (dict "description" .) -}}{{- end -}} {{- with .Params.tags -}}{{- $page = merge $page (dict "keywords" .) -}}{{- end -}} {{- /* No articleBody. The previous version inlined the entire rendered page body into every BlogPosting block, which inflated the HTML of every page on the site for a property Google does not consume. */ -}} {{- end -}} {{- /* ── site-wide entity (home only) ─────────────────────────────────── */ -}} {{- if .IsHome -}} {{- $schemaParams := site.Params.schema | default dict -}} {{- $entityType := $schemaParams.type | default "Organization" -}} {{- $entity := dict "@context" "https://schema.org" "@type" $entityType "url" site.BaseURL -}} {{- /* $authorName is params.author.name falling back to site.Title. Turning the visible trail off must stop claiming a breadcrumb in markup, and a top-level page that renders no trail must not describe one either. ONE list per page. shows. */ -}} {{- $showBreadCrumbs := .Param "showBreadCrumbs" | default true -}} {{- if and (not .IsHome) $showBreadCrumbs (gt (len .Ancestors) 1) -}} {{- $items := slice (dict "@type" "ListItem" "position" 1 "name" "Home" "item" site.BaseURL) -}} {{- $pos := 1 -}} {{- range after 1 .Ancestors.Reverse -}} {{- $pos = add $pos 1 -}} {{- $items = $items | append (dict "@type" "ListItem" "position" $pos "name" .LinkTitle "item" .Permalink) -}} {{- end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $pos 1) "name" .Title) -}} {{- $crumbs := dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $items -}} {{- end -}}
