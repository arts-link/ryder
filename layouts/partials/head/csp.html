{{- /* Content Security Policy (CSP) meta tag. Builds a CSP in enforcement mode from a base policy and optional per-site additions configured under [params.csp] in hugo.toml. Because Hugo generates static HTML the policy is delivered via a tag, which is universally supported without requiring server configuration. Default policy covers all resources that the theme itself loads: • Self-hosted (fingerprinted) CSS, JS and fonts • Google Fonts (googleapis.com / gstatic.com) • Plausible, PostHog, and Google Analytics when configured Users can extend each directive by setting extra sources in hugo.toml: [params.csp] imgSrc = "https://cdn.example.com" scriptSrc = "https://cdn.example.com" styleSrc = "https://cdn.example.com" connectSrc = "https://api.example.com" fontSrc = "https://cdn.example.com" frameSrc = "https://your-embed-host.example.com" embeds = ["youtube", "vimeo", "soundcloud", "spotify", "umap"] extraDirectives = "worker-src 'none';" frame-src is assembled from three sources, folded together and deduped: 1. Hosts the theme's own embed shortcodes (soundcloud, openstreetmap) register automatically on the page's .Store when used on the page. 2. params.csp.embeds — a list of preset names mapped to known hosts, for embeds the theme cannot auto-detect (e.g. Hugo's built-in youtube/vimeo shortcodes). 3. params.csp.frameSrc — any additional hosts, verbatim. If none of the three produce anything, frame-src is omitted from the policy entirely, same as before this existed. Set params.csp.disabled = true to opt out of the meta CSP tag entirely (e.g. when the hosting platform delivers CSP via HTTP headers instead). */ -}} {{- $cspParams := site.Params.csp | default dict -}} {{- if not $cspParams.disabled -}} {{- /* ── script-src ─────────────────────────────────────────────────────── */ -}} {{- $scriptSrc := slice "'self'" -}} {{- if not hugo.IsProduction -}} {{- $scriptSrc = $scriptSrc | append "'unsafe-inline'" "http://localhost:*" "http://127.0.0.1:*" -}} {{- end -}} {{- /* Plausible Analytics */ -}} {{- $plausibleDomain := site.Params.plausible_domain -}} {{- if $plausibleDomain -}} {{- $scriptSrc = $scriptSrc | append "https://plausible.io" -}} {{- /* SHA-256 hashes for Plausible advanced-mode inline scripts */ -}} {{- if site.Params.plausible_advanced -}} {{- $scriptSrc = $scriptSrc | append "'sha256-/6SBPqW+GW+//4nlXX6Y1nR9dWlh0gsQJ6KK71djH6A='" -}} {{- $scriptSrc = $scriptSrc | append "'sha256-RqYEbQ5LE8+gxvgfBOWbOoYTcxVwtSabsMrOzsDyWuQ='" -}} {{- end -}} {{- end -}} {{- /* PostHog Analytics */ -}} {{- $posthogProvider := eq (site.Params.analytics_provider | default "") "posthog" -}} {{- $posthogHost := site.Params.posthog_host | default "" -}} {{- if and $posthogProvider (not $posthogHost) -}} {{- /* getenv hard-errors the whole build if PUBLIC_POSTHOG_HOST isn't allowed by [security.funcs] getenv — see posthog.html, which warns about this same allowlist requirement. try keeps a missing/blocked host from taking down the entire CSP build. */ -}} {{- $rHost := try (getenv "PUBLIC_POSTHOG_HOST") -}} {{- if not $rHost.Err -}}{{- $posthogHost = $rHost.Value | default "" -}}{{- end -}} {{- end -}} {{- if and $posthogProvider $posthogHost -}} {{- /* PostHog's bootstrap snippet is inline, so allow inline scripts when active. */ -}} {{- $scriptSrc = $scriptSrc | append "'unsafe-inline'" -}} {{- $scriptSrc = $scriptSrc | append (replace $posthogHost ".i.posthog.com" "-assets.i.posthog.com") -}} {{- end -}} {{- /* Google Analytics (Hugo internal template uses googletagmanager.com) */ -}} {{- if and hugo.IsProduction (site.Config.Services.GoogleAnalytics.ID) -}} {{- $scriptSrc = $scriptSrc | append "https://www.googletagmanager.com" -}} {{- end -}} {{- with $cspParams.scriptSrc -}} {{- $scriptSrc = $scriptSrc | append . -}} {{- end -}} {{- /* ── style-src ─────────────────────────────────────────────────────── */ -}} {{- /* 'unsafe-inline' is required by Alpine.js (x-show adds display:none as an inline style) and for any user-provided inline styles. */ -}} {{- $styleSrc := slice "'self'" "'unsafe-inline'" "https://fonts.googleapis.com" -}} {{- with $cspParams.styleSrc -}} {{- $styleSrc = $styleSrc | append . -}} {{- end -}} {{- /* ── font-src ──────────────────────────────────────────────────────── */ -}} {{- $fontSrc := slice "'self'" "https://fonts.gstatic.com" -}} {{- with $cspParams.fontSrc -}} {{- $fontSrc = $fontSrc | append . -}} {{- end -}} {{- /* ── img-src ───────────────────────────────────────────────────────── */ -}} {{- /* Default restricts to self and data URIs (Font Awesome SVG). Sites that embed images from external CDNs or third-party services should add those hosts via params.csp.imgSrc in hugo.toml, e.g. imgSrc = "https://cdn.example.com https://images.unsplash.com" or use imgSrc = "https:" to allow all HTTPS image sources. */ -}} {{- $imgSrc := slice "'self'" "data:" -}} {{- /* Leaflet loads map tiles from OpenStreetMap — allow when loadLeaflet is set */ -}} {{- if .Param "loadLeaflet" -}} {{- $imgSrc = $imgSrc | append "https://*.tile.openstreetmap.org" -}} {{- end -}} {{- with $cspParams.imgSrc -}} {{- $imgSrc = $imgSrc | append . -}} {{- end -}} {{- /* ── frame-src ───────────────────────────────────────────────────── */ -}} {{- $frameSrc := slice -}} {{- /* Hugo renders (and this partial) before .Content, so any host an embed shortcode registers via .Store would otherwise arrive too late to read here. Force .Content evaluation now so shortcodes have already run. `with` no-ops safely on pages with no content (e.g. a section without an _index.md body) instead of erroring. */ -}} {{- with .Content -}}{{- end -}} {{- with .Store.GetSortedMapValues "cspFrameSrc" -}} {{- range . -}} {{- $frameSrc = $frameSrc | append . -}} {{- end -}} {{- end -}} {{- /* Known embed-host presets, for embeds the theme cannot auto-detect (e.g. Hugo's internal youtube/vimeo shortcodes, which have no hook to register themselves on .Store). */ -}} {{- $embedHostPresets := dict "youtube" (slice "https://www.youtube-nocookie.com" "https://www.youtube.com") "vimeo" (slice "https://player.vimeo.com") "soundcloud" (slice "https://w.soundcloud.com") "spotify" (slice "https://open.spotify.com") "umap" (slice "https://umap.openstreetmap.fr") -}} {{- range ($cspParams.embeds | default slice) -}} {{- with index $embedHostPresets . -}} {{- $frameSrc = $frameSrc | append . -}} {{- else -}} {{- warnf "params.csp.embeds: unknown preset %q (known: youtube, vimeo, soundcloud, spotify, umap)" . -}} {{- end -}} {{- end -}} {{- with $cspParams.frameSrc -}} {{- $frameSrc = $frameSrc | append . -}} {{- end -}} {{- $frameSrc = $frameSrc | uniq -}} {{- /* ── connect-src ───────────────────────────────────────────────────── */ -}} {{- $connectSrc := slice "'self'" -}} {{- if not hugo.IsProduction -}} {{- $connectSrc = $connectSrc | append "ws://localhost:*" "ws://127.0.0.1:*" "http://localhost:*" "http://127.0.0.1:*" -}} {{- end -}} {{- if $plausibleDomain -}} {{- $connectSrc = $connectSrc | append "https://plausible.io" -}} {{- end -}} {{- if and $posthogProvider $posthogHost -}} {{- $connectSrc = $connectSrc | append $posthogHost -}} {{- end -}} {{- if and hugo.IsProduction (site.Config.Services.GoogleAnalytics.ID) -}} {{- $connectSrc = $connectSrc | append "https://www.google-analytics.com" "https://region1.google-analytics.com" -}} {{- end -}} {{- with $cspParams.connectSrc -}} {{- $connectSrc = $connectSrc | append . -}} {{- end -}} {{- /* ── assemble policy string ───────────────────────────────────────── */ -}} {{- $policy := printf "default-src 'self'; script-src %s; style-src %s; font-src %s; img-src %s; connect-src %s;%s frame-ancestors 'none'; form-action 'self'; base-uri 'self'; object-src 'none';" (delimit $scriptSrc " ") (delimit $styleSrc " ") (delimit $fontSrc " ") (delimit $imgSrc " ") (delimit $connectSrc " ") (cond (gt (len $frameSrc) 0) (printf " frame-src %s;" (delimit $frameSrc " ")) "") -}} {{- with $cspParams.extraDirectives -}} {{- $policy = printf "%s %s" $policy . -}} {{- end -}} {{- end -}}
