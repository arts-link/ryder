{{- /* Favicon links. Configurable via [params.favicon] in site config — NOT via front matter / .Param: head.html calls this partial through partialCached with no explicit cache key, so it renders once and that single render is reused verbatim for every page. A per-page override would silently apply to every OTHER page too, which is worse than no override at all — so this reads site.Params directly rather than .Param, to make that constraint impossible to miss by copying the wrong pattern from elsewhere in the theme. [params.favicon] ico = "/favicon.ico" # default; theme ships this file version = "4" # cache-busting ?v= suffix; "" to omit svg = "/images/favicon.svg" # default; theme ships this file appleTouchIcon = "/images/apple-touch-icon.png" # default; theme ships this file webmanifest = "" # no default; theme ships no manifest file Set any of these to "" explicitly to omit that tag entirely. Sized PNG icons — the `favicon-32x32.png` / `favicon-16x16.png` pair most favicon generators emit — have no default and are opt-in, since the theme ships no such files. Each entry needs `href`; `size` is optional and, when present, is emitted as both the `sizes` attribute and nothing else: [[params.favicon.icons]] size = "32x32" href = "/favicon-32x32.png" [[params.favicon.icons]] size = "16x16" href = "/favicon-16x16.png" */ -}} {{- $favicon := site.Params.favicon | default dict -}} {{- $icoPath := $favicon.ico -}} {{- if not (isset $favicon "ico") -}}{{- $icoPath = "/favicon.ico" -}}{{- end -}} {{- $version := $favicon.version -}} {{- if not (isset $favicon "version") -}}{{- $version = "4" -}}{{- end -}} {{- $svgPath := $favicon.svg -}} {{- if not (isset $favicon "svg") -}}{{- $svgPath = "/images/favicon.svg" -}}{{- end -}} {{- /* NOTE: Hugo lower-cases config keys, so every isset lookup here must use the lower-cased form. "appleTouchIcon" never matches, which silently discards the site's value and falls back to the default. */ -}} {{- $appleTouchIcon := $favicon.appletouchicon -}} {{- if not (isset $favicon "appletouchicon") -}}{{- $appleTouchIcon = "/images/apple-touch-icon.png" -}}{{- end -}} {{- $webmanifest := $favicon.webmanifest -}} {{- with $icoPath -}} {{- $icoURL := . | relURL -}} {{- with $version -}}{{- $icoURL = printf "%s?v=%s" $icoURL . -}}{{- end -}} {{- end -}} {{- with $svgPath -}} {{- end -}} {{- range $favicon.icons -}} {{- $size := .size -}} {{- with .href -}} {{- end -}} {{- end -}} {{- with $appleTouchIcon -}} {{- end -}} {{- with $webmanifest -}} {{- end -}}
