{{- /* Video lightbox: a clickable thumbnail that opens a modal with the video embedded, beside main.js's existing imageGallery (images only). Uses the videoLightbox Alpine.data() component registered in main.js — CSP-safe: no inline calls, everything the handler needs travels as a data attribute or lives on the component. The iframe's src is only set once the modal opens (videoLightboxEmbedUrl starts empty), so nothing is requested from the embed host until the visitor actually clicks. Registers its provider's iframe host on .Page.Store, same as soundcloud/openstreetmap/youtube-embed/spotify-embed, so head/csp.html folds it into frame-src automatically. @param {string} provider "youtube" or "vimeo". Defaults to "youtube". @param {string} id Video ID for that provider (required). @param {string} thumbnail Thumbnail image: a page-bundle resource name, or an assets/-relative path. Falls back to a plain play-button tile with no image if omitted. @param {string} title Accessible label for the trigger and iframe title. Defaults to "Play video". @example: {{< video-lightbox provider="youtube" id="VLvVNMbQIRY" thumbnail="video-thumb.jpg" >}} */ -}} {{- $id := .Get "id" -}} {{- if not $id -}} {{- errorf "video-lightbox shortcode requires an id param" -}} {{- end -}} {{- $provider := .Get "provider" | default "youtube" -}} {{- $embedHosts := dict "youtube" "https://www.youtube-nocookie.com" "vimeo" "https://player.vimeo.com" -}} {{- $host := index $embedHosts $provider -}} {{- if not $host -}} {{- errorf "video-lightbox shortcode: unknown provider %q (known: youtube, vimeo)" $provider -}} {{- end -}} {{- .Page.Store.SetInMap "cspFrameSrc" $host $host -}} {{- $embedUrl := "" -}} {{- if eq $provider "youtube" -}} {{- $embedUrl = printf "https://www.youtube-nocookie.com/embed/%s?autoplay=1" $id -}} {{- else -}} {{- $embedUrl = printf "https://player.vimeo.com/video/%s?autoplay=1" $id -}} {{- end -}} {{- $title := .Get "title" | default "Play video" -}} {{- $thumbSrc := "" -}} {{- with .Get "thumbnail" -}} {{- $thumbPath := . -}} {{- with $.Page.Resources.Get $thumbPath -}} {{- $thumbSrc = .RelPermalink -}} {{- else -}} {{- with resources.Get (strings.TrimPrefix "/" $thumbPath) -}} {{- $thumbSrc = .RelPermalink -}} {{- else -}} {{- $thumbSrc = $thumbPath -}} {{- end -}} {{- end -}} {{- end -}}
